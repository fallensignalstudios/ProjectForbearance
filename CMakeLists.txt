cmake_minimum_required(VERSION 3.20)
project(SovereignCallExpansion LANGUAGES CXX VERSION 0.1.0)

# E0: engine-independent C++20 core with a headless command-line host. The core
# builds with no engine installed and no third-party dependency (TDD 3.2, 19.4).
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

option(EXPANSION_WERROR "Treat warnings as errors" ON)
option(EXPANSION_BUILD_TESTS "Build the verification suite" ON)

if(MSVC)
  add_compile_options(/W4 /permissive- /Zc:preprocessor)
  if(EXPANSION_WERROR)
    add_compile_options(/WX)
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion)
  if(EXPANSION_WERROR)
    add_compile_options(-Werror)
  endif()
endif()

set_target_properties(expansion_cli PROPERTIES OUTPUT_NAME expansion)

# ---------------------------------------------------------------------------
# Single-translation-unit build
#
# UnrealBuildTool compiles the .cpp files it discovers under a module directory.
# The simulation lives outside any module, so the Unreal host compiles it through
# one generated translation unit that includes each source in turn
# (docs/decisions/0010-unreal-host.md). That arrangement is not free: two static
# helpers with the same name in different files become an ambiguity, and a macro
# one file defines reaches the next. Building it here, with the same strict
# warnings, is what keeps the engine path buildable without an engine.
# ---------------------------------------------------------------------------
option(EXPANSION_BUILD_AMALGAMATION "Build the simulation as one translation unit" ON)

if(EXPANSION_BUILD_AMALGAMATION)
  set(EXPANSION_AMALGAMATION "${CMAKE_CURRENT_BINARY_DIR}/expansion_amalgamated.cpp")
  set(EXPANSION_AMALGAMATION_SOURCES
    ${EXPANSION_CORE_SOURCES} ${EXPANSION_PERSISTENCE_SOURCES} ${EXPANSION_PRESENTATION_SOURCES})
  set(_amalgam_body "// Generated by CMake. Do not edit; see EXPANSION_BUILD_AMALGAMATION.\n")
  foreach(_src IN LISTS EXPANSION_AMALGAMATION_SOURCES)
    string(APPEND _amalgam_body "#include \"${CMAKE_CURRENT_SOURCE_DIR}/${_src}\"\n")
  endforeach()
  file(GENERATE OUTPUT "${EXPANSION_AMALGAMATION}" CONTENT "${_amalgam_body}")

  add_library(expansion_amalgamated STATIC "${EXPANSION_AMALGAMATION}")
  target_include_directories(expansion_amalgamated PUBLIC
    core/include persistence/include presentation/include)
  # The sources include one another's private headers by path relative to the
  # repository root, as they do in the per-file build.
  target_include_directories(expansion_amalgamated PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
